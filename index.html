async function generatePitch() {

  const company = document.getElementById('p-company').value.trim();

  const role = document.getElementById('p-role').value.trim();

  if (!company || !role) return;

  const btn = document.getElementById('p-btn');

  const loading = document.getElementById('p-loading');

  const result = document.getElementById('p-result');

  btn.disabled = true; btn.textContent = 'Generating...';

  loading.classList.add('show'); result.classList.remove('show');

  try {

    const res = await fetch('https://api.anthropic.com/v1/messages', {

      method:'POST', headers:{'Content-Type':'application/json'},

      body: JSON.stringify({

        model:'claude-sonnet-4-20250514', max_tokens:1000,

        messages:[{role:'user',content:`Write a confident, personalized 3–4 sentence recruiter pitch about Subranil (Neil) Saha for a hiring manager at ${company} looking for a ${role}.\n\nAbout: Senior Android Engineer, 6+ years, Bengaluru. Kotlin, Jetpack Compose, KMM, MVVM, Bluetooth/IoT, Dagger. Built GroomTribe & OneBlade for Philips (IoT, KMM, AR) at UST Global; rebuilt PublicVibe (10M+) in KMM+Compose at Recro; led DIKSHA (1Cr+, Govt of India) at Tarento. 95%+ code coverage. Multi-module architecture expert.\n\nPunchy, specific, end with call to action. Pitch only.`}]

      })

    });

    const data = await res.json();

    const text = data.content?.map(b=>b.text||'').join('') || 'Could not generate pitch.';

    document.getElementById('p-text').textContent = text;

    result.classList.add('show');

  } catch(e) { document.getElementById('p-text').textContent='Something went wrong. Try again.'; result.classList.add('show'); }

  loading.classList.remove('show');

  btn.disabled = false; btn.textContent = 'Generate pitch ✦';

}
